{{- /* Return the Substack posts from data/substack.json as a slice of dicts. Substack blocks GitHub Actions, so the feed is snapshotted locally with scripts/update_substack.py instead of fetched at build time. */ -}} {{- $posts := slice }} {{- range site.Data.substack }} {{- $posts = $posts | append (dict "title" .title "link" .link "date" (time.AsTime .date) "description" .description ) }} {{- end }} {{- return $posts }}
